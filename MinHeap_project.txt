class MinHeap:
    def __init__(self):
        self.heap = []

    def push(self, item):
        self.heap.append(item)
        self._sift_up(len(self.heap) - 1)

    def pop(self):
        if not self.heap:
            raise IndexError("Heap is empty")
        if len(self.heap) == 1:
            return self.heap.pop()
        
        result = self.heap[0]
        self.heap[0] = self.heap.pop()
        if self.heap:
            self._sift_down(0)
        return result

    def peek(self):
        if not self.heap:
            raise IndexError("Heap is empty")
        return self.heap[0]

    def _sift_up(self, index):
        parent = (index - 1) // 2
        if index > 0 and self.heap[index][0] < self.heap[parent][0]:
            self.heap[index], self.heap[parent] = self.heap[parent], self.heap[index]
            self._sift_up(parent)

    def _sift_down(self, index):
        smallest = index
        left = 2 * index + 1
        right = 2 * index + 2

        if left < len(self.heap) and self.heap[left][0] < self.heap[smallest][0]:
            smallest = left

        if right < len(self.heap) and self.heap[right][0] < self.heap[smallest][0]:
            smallest = right

        if smallest != index:
            self.heap[index], self.heap[smallest] = self.heap[smallest], self.heap[index]
            self._sift_down(smallest)

    def is_empty(self):
        return len(self.heap) == 0

def minimize_max_cpu_load(jobs):
    if not jobs:
        return 0

    # Sort jobs by start time
    jobs.sort(key=lambda x: x[0])
    
    heap = MinHeap()
    current_load = 0
    max_load = 0

    for start, end, load in jobs:
        # Remove all jobs that end before current job starts
        while not heap.is_empty() and heap.peek()[0] <= start:
            _, finished_load = heap.pop()
            current_load -= finished_load

        # Add current job
        current_load += load
        heap.push((end, load))
        
        # Update max_load if current_load is greater
        max_load = max(max_load, current_load)

    return max_load

def validate_input(n, jobs):
    if not isinstance(n, int) or n < 0:
        raise ValueError("Number of jobs must be a non-negative integer")
    
    if len(jobs) != n:
        raise ValueError(f"Expected {n} jobs, but got {len(jobs)}")
    
    for start, end, load in jobs:
        if not all(isinstance(x, int) for x in (start, end, load)):
            raise ValueError("All job values must be integers")
        if start < 0 or end < 0 or load < 0:
            raise ValueError("Job values cannot be negative")
        if end <= start:
            raise ValueError("End time must be greater than start time")

def main():
    try:
        # Read input
        with open('inputPS06.txt', 'r') as f:
            n = int(f.readline().strip())
            jobs = []
            for _ in range(n):
                start, end, load = map(int, f.readline().strip().split())
                jobs.append((start, end, load))

        # Validate input
        validate_input(n, jobs)

        # Process jobs
        result = minimize_max_cpu_load(jobs)

        # Write output
        with open('outputPS06.txt', 'w') as f:
            f.write(f"Minimum Max CPU Load: {result}\n")

    except FileNotFoundError:
        print("Error: Input file 'inputPS06.txt' not found")
    except ValueError as e:
        print(f"Error: {str(e)}")
    except Exception as e:
        print(f"Error: An unexpected error occurred - {str(e)}")

if __name__ == "__main__":
    main()
